  Apply a basic static Cache in the release namespace:

  name: cache-sample
  namespace: {{ .Release.Namespace }}
spec:
  scaling:
    static:
      numServers: 2
  serverSizeInGB: 1000
  serverThreads: 80
  port: 9065
  metricsPort: 9096
  cpuLimit: 4
  memoryLimitMB: 128000
  nodeSelectorKey: dacs.azure.com/processing-unit

    evictionMaintainFreeSpace: true
    lastAccessThreshold: 30d
    passFrequency: 15m
    sizeTarget: "80"
    sizeThreshold: "90"
    bucketInterval: 15m
  storage:
    hostPath:
      nodeCacheFolderPath: /var/lib/ssd/cacheserver
