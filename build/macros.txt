;= ********************************************************************************
;= * Macros file for DACS
;= *
;= * Macros in this file are shared between cmd.exe and PowerShell, so anything
;= * in this file needs to work in both environments. If you have a macro which is
;= * specific to one environment or the other, put it in the corresponding
;= * shell-specific file next to this one.
;= *
;= * WHEN YOU SAVE THIS FILE PLEASE MAKE SURE THAT A UTF-8 BOM IS NOT ADDED.
;= *
;= * Be wary of using semicolons in macros, as PowerShell will interpret that as a
;= * separation between commands.
;= *
;= ********************************************************************************

;= ********************************************************************************
;= * Build macros (No configuration specified so Debug is implied)
;= ********************************************************************************
b=msbuild /m /v:minimal /clp:Summary /restore $*
bs=msbuild /m /v:minimal /clp:Summary /restore /p:SKIP_NATIVE_BUILD=TRUE $*
bo=msbuild /m:1 /v:minimal /p:BuildProjectReferences=false $*
bc=msbuild /m /v:minimal /clp:Summary /restore /t:Rebuild $*
bco=msbuild /m:1 /v:minimal /t:Rebuild /p:BuildProjectReferences=false $*
;= Native builds
nut=msbuild /m /v:minimal /clp:Summary /t:NativeUnitTest /restore $*

;= ********************************************************************************
;= * Build macros (Release configuration)
;= ********************************************************************************
br=msbuild /m /v:minimal /clp:Summary /p:Configuration=Release /restore $*
bor=msbuild /m:1 /v:minimal /p:BuildProjectReferences=false /p:Configuration=Release $*
bcr=msbuild /m /v:minimal /clp:Summary /t:Rebuild /p:Configuration=Release /restore $*
bcor=msbuild /m:1 /v:minimal /t:Rebuild /p:BuildProjectReferences=false /p:Configuration=Release $*
;= Native builds
nutr=msbuild /m /v:minimal /clp:Summary /p:Configuration=Release /t:NativeUnitTest /restore $*

;= ********************************************************************************
;= * Build macros (with logging)
;= ********************************************************************************
bl=msbuild /m "/flp1:LogFile=msbuild.log;PerformanceSummary;Summary;ShowTimestamp" "/flp2:LogFile=msbuild.wrn;WarningsOnly" "/flp3:LogFile=msbuild.err;ErrorsOnly" /restore $*
bcl=msbuild /m /t:Rebuild "/flp1:LogFile=msbuild.log;PerformanceSummary;Summary;ShowTimestamp" "/flp2:LogFile=msbuild.wrn;WarningsOnly" "/flp3:LogFile=msbuild.err;ErrorsOnly" /restore $*

;= ********************************************************************************
;= * Navigation aids
;= ********************************************************************************
up=cd ..
up1=cd ..
up2=cd ..\..
up3=cd ..\..\..
up4=cd ..\..\..\..
up5=cd ..\..\..\..\..
up6=cd ..\..\..\..\..\..
up7=cd ..\..\..\..\..\..\..
src=pushd "%BUILDDIR%"
bindir=pushd "%BINARIESDIR%"
reliability=cd %BUILDDIR%\subsystems\reliability

;= ********************************************************************************
;= * Unit tests
;= ********************************************************************************
ut=%PIPELINESDIR%\unittest.cmd
utr=%PIPELINESDIR%\unittest.cmd Release

;= ********************************************************************************
;= * Container Registry
;= ********************************************************************************
;= Login to Azure Container Registry (needed for container builds)
acrl=az acr login -n dacs

;= ********************************************************************************
;= * Open in Visual Studio
;= ********************************************************************************
open=msbuild /clp:Summary /t:SlnGen -Restore $*

n=notepad.exe $*
e=explorer $*
;= List macros
lm=doskey /macros
